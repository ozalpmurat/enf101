% ---- KENAR ÇUBUĞU ve KAPAK BANDI ----
% Bant, kenar çubuğunun ARKA PLANINI çizen şablondan basılır: beamer bu katmanı
% her derlemede güvenilir biçimde çizer. (DİKKAT: daha önce bant, kapak
% şablonunda bir tikz overlay olarak çiziliyordu; sayfa çıkışında dizildiği için
% derleme geçişine göre bir var bir yok oluyordu — bu yöntem terk edildi.)
%   · kapakta (1. slayt) → 3 cm genişliğinde bant
%   · diğer slaytlar     → kenar çubuğu genişliğinde bant (üzerine bölüm listesi gelir)
% Kapak ölçütü SLAYT NUMARASIDIR, bölüm sayacı DEĞİL: 1. haftada ilk '#' bölümünden
% önce bir '##' slaytı var; bölüm sayacı orada da 0 kaldığı için o slayt yanlışlıkla
% kapak sayılıyordu (3 cm'lik bant madde imlerini örtüyordu).
\makeatletter
\setbeamertemplate{sidebar canvas left}{%
  \ifnum\insertframenumber=1
    \color{bseured}\vrule height\sidebarheight width 3.0cm
  \else
    \ifbeamercolorempty[bg]{sidebar left}{}%
      {\color{bg}\vrule height\sidebarheight width\beamer@leftsidebar}%
  \fi}
% Kenar çubuğu: kapağa basılmaz; içerik slaytlarında sunu adı ve ders adı
% YAZILMAZ, onların yerine beyaz amblem (logobeyaz.png) durur — amblem şeffaflığı
% korunarak banda basılır. Sunu adı zaten her slaytın alt şeridinde yazıyor.
% Altında bölüm navigasyonu aynen kalır.
\setbeamertemplate{sidebar left}{%
  \ifnum\insertframenumber>1
    \beamer@tempdim=\beamer@sidebarwidth%
    \advance\beamer@tempdim by -0.4cm%
    \vskip1.2em%
    \hbox to \beamer@sidebarwidth{\hss
      \includegraphics[width=\beamer@tempdim]{../tema/logobeyaz.png}\hss}%
    \vskip1.4em%
    \insertverticalnavigation{\beamer@sidebarwidth}%
    \vfill
  \fi}
\makeatother

% ---- KAPAK METNİ + LOGO ----
% Logo, bandın sağ kenarına (3,0 cm) ortalanır; metin alanının sol kenarı
% sayfadan (kenar çubuğu + metin marjı) kadar içeride olduğu için fark
% \dimexpr ile hesaplanır. Logo içerik olarak basılır (kırpılmaz).
\makeatletter
\setbeamertemplate{title page}{%
  \begingroup
  % Kapakta sözcük BÖLÜNMEZ ve satırlar ESNEMEZ: metin sola dayalıdır.
  % (Aksi hâlde başlık "Okuryazar-" / "lık" diye tirenle bölünüyor ya da
  %  iki yana yaslamada sözcükler arası kocaman boşluklar açılıyor.)
  \hyphenpenalty=10000 \exhyphenpenalty=10000
  \vspace*{0.75cm}%
  \hspace*{3.6cm}%
  \begin{minipage}{9.2cm}
    \raggedright
    {\color{bseutint}\rule{2.6cm}{1.4pt}\par}
    \vspace{4mm}
    {\color{bseured}\LARGE\bfseries \inserttitle\par}
    \vspace{5mm}
    {\color{bseugrey}\large \insertsubtitle\par}
    \vspace{6mm}
    {\color{bseugrey}\small \insertauthor\par}
    {\color{bseugrey}\small Bilecik Şeyh Edebali Üniversitesi\par}
  \end{minipage}
  \vfill
  {\hspace*{\dimexpr 1.5cm-\beamer@leftsidebar-\beamer@leftmargin\relax}%
   \includegraphics[height=3cm]{../tema/logo.png}\par}%
  \endgroup
}
\makeatother

% ---- ALT ŞERİT: sunu adı + sayfa numarası (kapakta çizilmez) ----
\setbeamertemplate{footline}{%
  \ifnum\insertframenumber>1
    \begin{beamercolorbox}[colsep=1.5pt]{upper separation line foot}\end{beamercolorbox}%
    \begin{beamercolorbox}[ht=2.5ex,dp=1.125ex,leftskip=.3cm,rightskip=.3cm plus1fil]{title in head/foot}%
      \leavevmode{\usebeamerfont{title in head/foot}\insertshorttitle}%
      \hfill%
      {\usebeamerfont{institute in head/foot}\usebeamercolor[fg]{institute in head/foot}\insertshortinstitute
       \qquad\usebeamercolor[fg]{page number in head/foot}\usebeamerfont{page number in head/foot}\insertframenumber\,/\,\inserttotalframenumber}%
    \end{beamercolorbox}%
    \begin{beamercolorbox}[colsep=1.5pt]{lower separation line foot}\end{beamercolorbox}%
  \fi}
